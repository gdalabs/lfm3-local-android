cmake_minimum_required(VERSION 3.31.6)
project(meshmind LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS meshmind_compile=2 meshmind_link=1)
set(CMAKE_JOB_POOL_COMPILE meshmind_compile)
set(CMAKE_JOB_POOL_LINK meshmind_link)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(ANDROID)
    if(ANDROID_ABI STREQUAL "x86_64" AND MESHMIND_EMULATOR)
        set(GGML_AVX OFF CACHE BOOL "" FORCE)
        set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
        set(GGML_FMA OFF CACHE BOOL "" FORCE)
        set(GGML_F16C OFF CACHE BOOL "" FORCE)
    elseif(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "Phase one supports arm64-v8a only")
    else()
        set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
    endif()
endif()
add_subdirectory(../third_party/llama.cpp upstream)
add_library(meshmind-runtime STATIC runtime.cpp)
target_include_directories(meshmind-runtime PUBLIC .)
target_link_libraries(meshmind-runtime PUBLIC llama llama-common)
option(MESHMIND_RUNTIME_SANITIZERS "Instrument the host adapter (upstream remains uninstrumented)" OFF)
if(MESHMIND_RUNTIME_SANITIZERS AND NOT ANDROID)
    target_compile_options(meshmind-runtime PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(meshmind-runtime INTERFACE -fsanitize=address,undefined)
endif()
if(ANDROID)
    add_library(meshmind SHARED jni.cpp)
    target_link_libraries(meshmind PRIVATE meshmind-runtime android log)
    target_link_options(meshmind PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(meshmind-bench ../benchmark/main.cpp)
    target_link_libraries(meshmind-bench PRIVATE meshmind-runtime)
    add_executable(lfm3-runtime-test ../tests/runtime_integration.cpp)
    target_link_libraries(lfm3-runtime-test PRIVATE meshmind-runtime)
endif()
